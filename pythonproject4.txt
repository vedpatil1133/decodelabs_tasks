score = 0

# Question 1
answer = input("Q1. What is the capital of France? ").strip().lower()

if answer == "paris":
    print("Correct!")
    score += 1
else:
    print("Wrong!")

# Question 2
answer = input("Q2. Which planet is known as the Red Planet? ").strip().lower()

if answer == "mars":
    print("Correct!")
    score += 1
else:
    print("Wrong!")

# Question 3
answer = input("Q3. How many days are there in a week? ").strip().lower()

if answer == "7" or answer == "seven":
    print("Correct!")
    score += 1
else:
    print("Wrong!")

# Final score
print(f"\nYour final score is: {score}/3")
